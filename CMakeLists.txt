cmake_minimum_required(VERSION 3.10)
project(collision_avoidance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
option(BUILD_ROS "Build the ROS 1 adapter (requires catkin)" ON)
include_directories(include)
if(BUILD_ROS)
  find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs sensor_msgs tf laser_geometry)
  catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS roscpp geometry_msgs sensor_msgs tf laser_geometry)
  include_directories(${catkin_INCLUDE_DIRS})
  add_executable(SimNode src/SimNode.cpp)
  target_link_libraries(SimNode ${catkin_LIBRARIES})
  install(TARGETS SimNode RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
  install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
  install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(controller_tests tests/controller_tests.cpp)
  add_test(NAME controller COMMAND controller_tests)
endif()
add_executable(simulate tools/simulate.cpp)

if(BUILD_TESTING)
  foreach(scenario single offset multiple clear)
    add_test(NAME scenario_${scenario} COMMAND simulate ${CMAKE_CURRENT_BINARY_DIR}/${scenario}.csv ${scenario})
  endforeach()
endif()
